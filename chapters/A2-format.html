Appendix A2. Format specifier reference

This appendix is the complete table of what can go inside {…} in a format string for print, allocPrint, bufPrint and std.debug.print in Zig 0.16. Every example was printed by one program (a2_fmt.zig, compiled and run while writing this page) and the output column is what it printed. If a placeholder is not in this table, it does not exist in 0.16.

A2.1 The grammar

A placeholder is

{[argument][specifier]:[fill][alignment][width].[precision]}
  • argument: nothing (the next argument in order), a number ({0}, {1}), or a field name in square brackets ({[score]}) when the argument tuple is a struct literal with named fields.
  • specifier: one of the letters in A2.2, chosen by the type of the argument.
  • fill: one byte used for padding. If you give a fill you must also give an alignment, or the byte is read as a width.
  • alignment: < left, ^ centre, > right.
  • width: the minimum field width in bytes. Alignment is by bytes, not by displayed characters; it is only right for ASCII.
  • precision: for floats, the number of decimals; for {s}, a maximum length.

A literal brace is doubled: {{ prints { and }} prints }. That matters whenever a format string contains IRON LEDGER's {g}…{/} colour markup (Chapter 44).

Things that do not work. {} on a slice or string is a compile error ("cannot format slice without a specifier (i.e. {s}, {x}, {b64}, or {any})"): you must say {s} or {any}. {d} on an array or slice of integers is an error too; use {any}. {D} (nanoseconds as a duration) is mentioned in the standard library's doc comment but is rejected for integers in 0.16.0 ("invalid format string 'D' for type 'u64'"); do not use it. And the old format(self, comptime fmt, options, writer: anytype) method signature is no longer called by {}; see A3.

The IRON LEDGER row format from Chapter 44 uses these with a space after the colon, {d: <4}, which reads as "fill with space, left, width 4"; the space is the fill byte, and it is the same as {d:<4}.

A2.4 The program

For reference, the file that produced every output above. Copy it into your scratch directory and run it with zig run; it is a quick way to check a placeholder before using it.

const std = @import("std");
const Point = struct {
    x: i32,
    y: i32,
    pub fn format(self: Point, w: *std.Io.Writer) std.Io.Writer.Error!void {
        try w.print("({d}, {d})", .{ self.x, self.y });
    }
};
const Color = enum { red, green };
pub fn main(init: std.process.Init) !void {
    var buf: [4096]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &buf);
    const out = &w.interface;
    const n: i32 = -42;
    const f: f64 = 3.14159;
    const s: []const u8 = "iron";
    const opt: ?u8 = null;
    const opt2: ?u8 = 7;
    const eu: anyerror!u8 = error.Boom;
    const arr = [_]u8{ 1, 2, 3 };
    const p = Point{ .x = 1, .y = 2 };
    try out.print("{{}}       |{}|\n", .{n});
    try out.print("{{d}}      |{d}|\n", .{n});
    try out.print("{{d}} f    |{d}|\n", .{f});
    try out.print("{{d:.2}}   |{d:.2}|\n", .{f});
    try out.print("{{e}}      |{e}|\n", .{f});
    try out.print("{{x}}      |{x}|\n", .{@as(u32, 255)});
    try out.print("{{x:0>8}}  |{x:0>8}|\n", .{@as(u32, 255)});
    try out.print("{{c}}      |{c}|\n", .{@as(u8, 'A')});
    try out.print("{{u}}      |{u}|\n", .{@as(u21, 0x263A)});
    try out.print("{{s:*^8}}  |{s:*^8}|\n", .{s});
    try out.print("{{any}}    |{any}|\n", .{arr});
    try out.print("{{?}}      |{?}|  |{?}|\n", .{ opt, opt2 });
    try out.print("{{!}}      |{!}|\n", .{eu});
    try out.print("{{t}}      |{t}|\n", .{Color.green});
    try out.print("{{f}}      |{f}|\n", .{p});
    try out.print("{{B}}      |{B}|\n", .{@as(u64, 1536)});
    try out.print("{{b64}}    |{b64}|\n", .{"iron"});
    try out.print("{{1s}} {{0s}} |{1s} {0s}|\n", .{ "first", "second" });
    try out.print("{{[b]}}{{[a]}} |{[b]}{[a]}|\n", .{ .a = 1, .b = 2 });
    try out.print("{{d:>[1]}} |{d:>[1]}|\n", .{ 7, 5 });
    try out.flush();
}
